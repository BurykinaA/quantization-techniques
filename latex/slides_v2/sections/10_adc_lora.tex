\begin{frame}[shrink=20]{Post-ADC Residual LoRA: Breaking the Plateau}
  \begin{columns}[T]
    \begin{column}{0.50\textwidth}
      \textbf{Architecture:}
      \small
      \[
        y = \underbrace{\mathrm{ADC}_{\mathrm{frozen}}(x)}_{\text{PTQ path}} +\;
            \tfrac{\alpha}{r}\, B\!\left(A(x_{\mathbb{R}})\right)
      \]
      \normalsize
      \begin{itemize}
        \item Correction applied \textbf{after} ADC output
        \item Base model frozen in fp16; LoRA adapters in fp32
        \item Gradients never touch \texttt{round\_ste} / ADC clamp $\Rightarrow$ stable
        \item Trained with LM cross-entropy (+ optional KL teacher), 5 epochs
      \end{itemize}

      \vspace{0.15em}

      \end{alertblock}
