\begin{frame}{Motivation}

  \vspace*{\fill}

  {\Large\bfseries LLM inference is expensive in energy.}

  \vspace{1.2em}

  {\Large\bfseries Analog optical hardware is}
  \,{\Large\bfseries\color{myorange}${\sim}10^{3}\times$}\,
  {\Large\bfseries more energy-efficient.}

  \vspace*{\fill}

  \begin{tikzpicture}[remember picture, overlay]
    \node[anchor=south west, inner sep=0pt]
      at ([xshift=0.6cm, yshift=0.35cm]current page.south west) {%
      {\tiny\color{mygray}%
        \href{https://arxiv.org/abs/2308.05226}{Bandyopadhyay et al.,
        \textit{Single chip photonic deep neural network with accelerated
        training}, arXiv:2308.05226}}};
  \end{tikzpicture}

  \note{Я не утверждаю, что вся LLM будет оптической. Идея в том, что самая дорогая часть inference — линейная алгебра — может быть вынесена в оптический accelerator.}

\end{frame}
