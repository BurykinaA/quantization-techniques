      \textbf{Overhead on Llama-3.2-1B}\\
      \emph{($d{=}2048$, 16 blocks, $r{=}4$, all 7 proj.)}\\[0.4em]

      \renewcommand{\arraystretch}{1.2}

        training (1 GPU, 5 ep.) & $\sim$\SI{45}{min} \\
        inference MAC overhead & $<$\,0.5\% \\
        \bottomrule
      \end{tabular}

      \vspace{0.7em}
      \begin{alertblock}{Observations}
        \scriptsize
        \begin{itemize}
          \item Rank gain saturates fast: $r{=}1$ already gives 15.9.
          \item KL distillation adds $\sim$1.2 PPL on top of CE.
          \item Early layers matter most — skipping first 8 hurts more than last 8.
          \item Pre-ADC LoRA \emph{diverges} — its correction is quantised by $\delta$.
        \end{itemize}
      \end{alertblock}
