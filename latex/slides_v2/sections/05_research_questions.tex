\begin{frame}{Research Questions}

  \vfill
  {\normalsize\textbf{Goal.}}\;
  Adapt LLM post-training quantisation to fixed ADC quantisation
  in analog optical inference.

  \vspace{1.5em}
  \begin{description}
    \setlength\itemsep{1.2em}
    \item[\color{mybluedark}\textbf{RQ1 — ADC-aware PTQ}]\hfill\\
      How far can pure PTQ go when the ADC is included during calibration?

    \item[\color{mybluedark}\textbf{RQ2 — Beyond the PTQ plateau}]\hfill\\
      When pure PTQ stops improving, what is the smallest additional
      correction needed to recover the remaining ADC-induced loss?
  \end{description}
  \vfill

\end{frame}
