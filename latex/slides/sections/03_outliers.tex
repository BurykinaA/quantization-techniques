  {\footnotesize\centering
    \textbf{Quantise:}\enspace
    $\hat v = \mathrm{clip}\!\left(\lfloor v/s\rfloor,\,-q,\,q\right)$,\enspace
    $s={\max|v|}/{q}$,\enspace $q=2^{b-1}{-}1$
    \hspace{1.5em}
    \textbf{Dequantise:}\enspace $\tilde v = s\cdot\hat v$\par}
