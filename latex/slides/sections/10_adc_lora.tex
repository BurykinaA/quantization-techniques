\begin{frame}{Post-ADC Residual LoRA: Breaking the Plateau}
  \begin{columns}[T]
    \begin{column}{0.50\textwidth}
      \textbf{Architecture:}
      \[
        y = \underbrace{\text{frozen ADC}(x)}_{\text{PTQ path}} +\;
            \frac{\alpha}{r}\, B\bigl(A(x_{\mathbb{R}})\bigr)
      \]
      \begin{itemize}
        \item Correction applied \textbf{after} ADC output
        \item Base model frozen in fp16; LoRA adapters in fp32
        \item Gradients never touch \texttt{round\_ste} / ADC clamp $\Rightarrow$ stable
        \item Trained with LM cross-entropy (+ optional KL teacher), 5 epochs
      \end{itemize}

      \vspace{0.5em}
      \begin{alertblock}{\textcolor{myred}{\textbf{Pre-ADC LoRA failed}}}
        \texttt{pre\_adc\_r4\_down\_o}: bypass=123, ADC=105 — catastrophic divergence.\\
        Gradients through $\texttt{round\_ste}(\mathbf{W}{+}\Delta\mathbf{W})$ are too noisy
        from step 0.
      \end{alertblock}

      \vspace{0.5em}
      \textbf{Best result} (\texttt{r4\_all\_ce\_kl}):

          + LoRA (all 7, r=4, CE+KL) & \best{$\adcppl = 13.96$} \\
          \bottomrule
        \end{tabular}
      \end{center}
    \end{column}
