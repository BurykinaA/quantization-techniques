      \begin{center}
      \begin{tikzpicture}[
          every node/.style={font=\small},
          box/.style={draw, rounded corners=2pt, minimum width=1.6cm, minimum height=0.55cm, align=center},

        ]
        \node[box, fill=myblue!20]  (q)   {Quantize\\$x \to \hat{x}$};
        \node[box, fill=mygray!20, right=0.5cm of q]  (mvm) {Integer\\MVM};
        \node[box, fill=myorange!20, right=0.5cm of mvm] (adc) {ADC\\$\lfloor y / \delta \rfloor$};
        \node[box, fill=mygreen!20, right=0.5cm of adc] (dq)  {Dequant};
        \draw[arr] (q) -- (mvm);
        \draw[arr] (mvm) -- (adc);
        \draw[arr] (adc) -- (dq);
        \node[above=0.15cm of q, font=\footnotesize] {$x$};
        \node[above=0.15cm of mvm, font=\footnotesize] {$y_\mathrm{int}$};
        \node[above=0.15cm of adc, font=\footnotesize] {$z = y_\mathrm{int}/\delta$};
      \end{tikzpicture}
      \end{center}

      \vspace{0.5em}
      \textbf{Dead zone:} $|y_\mathrm{int}| < \delta \Rightarrow z = 0$ (ADC outputs nothing)\\[0.3em]
      $\delta = \dfrac{2 \cdot \text{tile\_in} \cdot 127^2}{2^{b_a} \cdot k}
               = \dfrac{2 \cdot 256 \cdot 127^2}{256 \cdot 16} \approx 2016$

      \vspace{0.5em}
      \textbf{Key metrics}
      \begin{itemize}
        \item $\bypass$ — PPL with ADC disabled (transform quality proxy)
        \item $\adcppl$ — PPL with full ADC pipeline (what matters)
        \item \texttt{dead\_rate} — fraction of channels with $z=0$
      \end{itemize}
