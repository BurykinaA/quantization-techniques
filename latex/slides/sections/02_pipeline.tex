\begin{frame}[shrink=15]{Analog MVM Pipeline: Tiling and the Hardware-Fixed Step}

  \vspace{-0.1em}
  \footnotesize
  \textbf{Tile} = $M{\times}N$ block of the weight matrix computed in one analog shot; $k$ analog columns are multiplexed into one ADC. A large weight matrix is split into tiles; every tile gets exactly one ADC read per output column.

  \vspace{0.6em}
  \begin{center}
  \resizebox{0.90\linewidth}{!}{%
  \begin{tikzpicture}[
      every node/.style={font=\footnotesize},
      db/.style={draw, rounded corners=2pt, fill=myblue!15,
                 minimum width=1.9cm, minimum height=0.65cm, align=center},
      mb/.style={draw, rounded corners=2pt, fill=mybluedark!18, text=mybluedark,
                 minimum width=2.4cm, minimum height=0.85cm, align=center},
      ab/.style={draw, rounded corners=2pt, fill=myorange!40,
                 minimum width=1.8cm, minimum height=0.65cm, align=center},
      outbox/.style={draw, rounded corners=2pt, fill=mygreen!20,
                     minimum width=1.2cm, minimum height=0.65cm, align=center},
      arr/.style={-{Latex[length=2mm]}, thick}
    ]
    \node (w)   at (0,  1.0) {$w$};
    \node[db] (qw) at (2.2, 1.0) {Digital\\quantiser};
    \node (wb)  at (4.2, 1.0) {$\hat w$};

    \node (x)   at (0, -1.0) {$x$};
    \node[db] (qx) at (2.2,-1.0) {Digital\\quantiser};
    \node (xb)  at (4.2,-1.0) {$\hat x$};

    \draw[arr] (w) -- (qw);   \draw[arr] (qw) -- (wb);
    \draw[arr] (x) -- (qx);   \draw[arr] (qx) -- (xb);

    \node[mb] (mvm) at (6.4, 0) {Analog IMC tile\\ $y{=}\sum_i \hat w_i\hat x_i$};
    \draw[arr] (wb) -- (mvm.north west);
    \draw[arr] (xb) -- (mvm.south west);

    \node[ab] (adc) at (9.2, 0) {ADC\\$\lfloor y/\delta\rfloor$};
    \draw[arr] (mvm) -- node[above, font=\tiny] {$y\!\in\!\mathbb{Z}$} (adc);

    \node[outbox] (yb) at (11.2, 0) {$\hat y$};
    \draw[arr] (adc) -- (yb);
  \end{tikzpicture}}
  \end{center}

  \vspace{0.5em}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \footnotesize
      \textbf{Hardware-fixed step size:}
      \[
        \boxed{\; \delta \;=\; \frac{2\,M\,q_x\,q_w}{2^{b_a}\,k} \;}
      \]
      \scriptsize
      $q_x = 2^{b_x}{-}1,\; q_w = 2^{b_w-1}{-}1$; $M$ = tile size; $b_a$ = ADC bits; $k$ = ADC parallelism (integer, set at tape-out).
    \end{column}
    \begin{column}{0.46\textwidth}
      \footnotesize\renewcommand{\arraystretch}{1.2}
      \textbf{Example} ($M{=}256,\; b_a{=}8,\; k{=}4$):\\
      \begin{tabular}{@{}l c@{}}
        \toprule
        INT8 ($q_x{=}q_w{=}127$)  & $\delta \approx \mathbf{8064.5}$ \\
        INT4 ($q_x{=}q_w{=}7$)    & $\delta \,=\, \mathbf{24.5}$ \\
        \bottomrule
      \end{tabular}

      \vspace{0.4em}
      \alert{\textbf{Dead zone}:} $|y|{<}\delta \;\Rightarrow\; \hat y{=}0$.
    \end{column}
  \end{columns}

\end{frame}
